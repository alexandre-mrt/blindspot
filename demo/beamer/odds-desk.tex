\begin{frame}{Motivation}
  \begin{itemize}
    \item x402 lets an agent pay for each API call with a stablecoin payment.
    \item Every payment exposes the payer, the payee, the amount and the time, to the seller and to anyone reading the chain.
    \item For an agent that buys market data, this sequence is its strategy: which signal it reads and when it acts.
  \end{itemize}
  \pause
  \vspace{1em}
  \textbf{Question.} Can we keep per-request pricing while removing the link between a request and the agent that paid for it?
\end{frame}

\begin{frame}{Setting and threat model}
  \begin{itemize}
    \item Agent $A$ buys data from a seller $S$, which also runs the mint.
    \item Settlement goes through Circle Gateway on Arc (USDC, gas paid in USDC).
    \item $S$ is honest but curious: it serves paid requests correctly and tries to link them to payers.
    \item Chain observers see every settled transfer.
    \item Out of scope: network metadata (IP address, timing).
  \end{itemize}
\end{frame}

\begin{frame}{Background}
  \textbf{x402.} The server answers \texttt{402} with payment requirements; the client retries with a signed payment payload; a facilitator verifies and settles it.
  \medskip

  \textbf{Gateway nanopayments.} The client deposits once on-chain, then signs off-chain authorizations that Circle settles in batches.
  \pause
  \medskip

  \textbf{Blind RSA signatures} (Chaum, 1983). With key $(N, e, d)$ and a hash $H$ onto $\mathbb{Z}_N$:
  \[
    b = H(s)\, r^{e} \bmod N, \qquad
    s' = b^{d} \bmod N, \qquad
    \sigma = s'\, r^{-1} \bmod N = H(s)^{d} \bmod N.
  \]
  The signer sees only $b$, which is uniformly distributed for a random $r$.
\end{frame}

\begin{frame}{Construction}
  \textbf{Issuance} (one public payment per batch of $k = 5$ credits)
  \begin{enumerate}
    \item $A$ draws serials $s_1,\dots,s_k$ and blinding factors $r_1,\dots,r_k$.
    \item $A$ sends $b_i = H(s_i)\,r_i^{e}$ to \texttt{POST /credits/issue}, paid with one Gateway nanopayment of \$0.01.
    \item $S$ returns $b_i^{d}$; $A$ unblinds and checks $\sigma_i^{e} = H(s_i)$.
  \end{enumerate}
  \pause
  \textbf{Redemption} (every request)
  \begin{enumerate}
    \setcounter{enumi}{3}
    \item $A$ pays with the payload $(s_i, \sigma_i)$ under the scheme \texttt{blind-credit}.
    \item $S$ accepts if $\sigma_i^{e} = H(s_i)$ and $s_i$ is unspent, then records $s_i$.
  \end{enumerate}
  Implemented against the x402 interfaces \texttt{SchemeNetworkClient}, \texttt{SchemeNetworkServer} and \texttt{FacilitatorClient}.
\end{frame}

\begin{frame}{Properties and limits}
  \begin{tabular}{@{}ll@{}}
    \toprule
    Property & Argument \\
    \midrule
    Unlinkability & $S$ saw only $b_i$ at issuance; $(s_i,\sigma_i)$ is new to it at redemption \\
    Unforgeability & one-more RSA assumption (Bellare et al., 2003) \\
    No double spending & spent-serial set checked in verify, written in settle \\
    Price privacy & every resource costs one credit \\
    \bottomrule
  \end{tabular}
  \bigskip

  \textbf{Limits.} The batch purchase is linkable to $A$. Timing and IP are not hidden. The mint key is kept in memory.
\end{frame}

\begin{frame}{Spending policy}
  The agent's policy runs in the x402 client hook \texttt{onBeforePaymentCreation}, so a refused payment is never signed.
  \medskip
  \begin{itemize}
    \item origin allowlist
    \item per-call cap of \$0.01 and a session budget of \$0.05
    \item human approval above \$0.004, rejected after 60 s without an answer
  \end{itemize}
\end{frame}

\begin{frame}{Evaluation: is the purchased signal informative?}
  Gemini crypto strike ladders settled in the 24 hours before submission: 84 events, 736 contracts. Each contract is priced 60 minutes before expiry from the settlement index only.
  \medskip

  \centering
  \small
  \begin{tabular}{@{}lcc@{}}
    \toprule
    Forecaster & Brier, all ($n=736$) & Brier, $p \in (0.1, 0.9)$ ($n=222$) \\
    \midrule
    Lognormal model, 2\,h realized volatility & \textbf{0.112} & \textbf{0.217} \\
    Spot already past the strike & 0.152 & 0.342 \\
    Constant 0.5 & 0.250 & 0.250 \\
    \bottomrule
  \end{tabular}
  \medskip

  \raggedright
  The model is overconfident above 80\% (predicted 0.97, realized 0.87). Gemini publishes no contract price history, so profit against the order book is measured forward only.
\end{frame}

\begin{frame}{Evaluation: cost}
  \begin{itemize}
    \item Redeeming a credit: one RSA verification with $e = 65537$, \textbf{0.063 ms} on one laptop core (about 16{,}000 redemptions per second).
    \item Settlement: one Gateway nanopayment per batch of five credits.
    \item Demonstration run: three feeds bought for \$0.01 and no on-chain transaction per request.
  \end{itemize}
\end{frame}

\begin{frame}{Implementation}
  \begin{itemize}
    \item Settlement: Circle Gateway nanopayments, \texttt{@circle-fin/x402-batching} 4.0 (\texttt{GatewayClient}, \texttt{GatewayEvmScheme}, \texttt{BatchFacilitatorClient}).
    \item Protocol: x402 v2, \texttt{@x402/core}, \texttt{@x402/hono} and \texttt{@x402/fetch} 2.31, with \texttt{spendControls}.
    \item Chain: Arc testnet (\texttt{eip155:5042002}), USDC as gas.
    \item Data: Gemini Prediction Markets REST API and Lukka index klines.
    \item Added: the \texttt{blind-credit} scheme, with a specification in the repository.
  \end{itemize}
  \medskip
  Code: \texttt{github.com/alexandre-mrt/odds-desk}
\end{frame}

\begin{frame}{Future work}
  \begin{itemize}
    \item Propose \texttt{blind-credit} to the x402 specification.
    \item Persistent mint keys with epochs and expiry.
    \item Make the batch purchase unlinkable, for example with a Gateway deposit from a fresh address per batch.
    \item Redemption through a relay to remove network metadata.
  \end{itemize}
\end{frame}
