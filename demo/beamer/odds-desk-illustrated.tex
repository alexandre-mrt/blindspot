\begin{frame}{Motivation}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \begin{itemize}
        \item With x402 an agent pays for each API call with a stablecoin payment.
        \item Each payment shows payer, payee, amount and time.
        \item For an agent buying market data, that history is its strategy.
      \end{itemize}
    \end{column}
    \begin{column}{0.42\textwidth}
      \centering
      \node{agent}{agent}\arr{\$0.005}\node{merchant}{seller}\\[2mm]
      \node[1cm]{whale}{anyone reading the chain}
    \end{column}
  \end{columns}
  \pause
  \vspace{0.8em}
  \textbf{Question.} Can the agent keep paying per request while the seller can no longer tell which agent made which request?
\end{frame}

\begin{frame}{Setting and threat model}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \begin{itemize}
        \item Agent $A$ buys data from seller $S$, who also runs the mint.
        \item Settlement: Circle Gateway on Arc, USDC as gas.
        \item $S$ is honest but curious.
        \item Out of scope: IP address and timing.
      \end{itemize}
    \end{column}

  \end{columns}
\end{frame}

\begin{frame}{Background}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \textbf{x402.} \texttt{402}, then a signed payment, then a facilitator settles it.
      \medskip

      \textbf{Gateway nanopayments.} One deposit on-chain, then off-chain authorizations settled in batches.
    \end{column}
    \begin{column}{0.42\textwidth}
      \begin{term}[\linewidth]
        \cmd{curl -i localhost:4021/api/feeds/edge}
        \out{HTTP/1.1 402 Payment Required}
        \out{accepts: exact (Gateway), blind-credit}
      \end{term}
    \end{column}
  \end{columns}
  \pause
  \medskip
  \textbf{Blind RSA signatures} (Chaum, 1983), key $(N,e,d)$, hash $H$ onto $\mathbb{Z}_N$:\hfill\sprite[1cm]{blinded}
  \[
    b = H(s)\, r^{e} \bmod N, \qquad s' = b^{d} \bmod N, \qquad \sigma = s'\, r^{-1} = H(s)^{d} \bmod N.
  \]
\end{frame}

\begin{frame}{Construction}
  \centering
  \node[1.2cm]{agent}{agent}\arr{$b_1..b_5$ + \$0.01}\node[1.2cm]{mint}{mint}\arr{$b_i^d$}\node[1.2cm]{credit}{5 credits}
  \raggedright
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \textbf{Issuance.} Blind five serials, pay one Gateway nanopayment, unblind, check $\sigma_i^e = H(s_i)$.
    \end{column}
    \begin{column}{0.45\textwidth}
      \begin{term}[\linewidth]
        \cmd{POST /credits/issue}
        \out{gateway: \$0.01 signed off-chain, no gas}
        \out{mint: signed 5 blinded values}
      \end{term}
    \end{column}
  \end{columns}
  \pause
  \vspace{2mm}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \textbf{Redemption.} Pay each request with $(s_i,\sigma_i)$ under the scheme \texttt{blind-credit}; $S$ checks the signature and that $s_i$ is unspent.
    \end{column}
    \begin{column}{0.45\textwidth}
      \begin{term}[\linewidth]
        \cmd{GET /api/feeds/edge  (blind-credit)}
        \out{200 OK   payer = anonymous}
      \end{term}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Properties and limits}
  \begin{columns}[T]
    \begin{column}{0.74\textwidth}
      \small
      \begin{tabular}{@{}ll@{}}
        \toprule
        Property & Argument \\
        \midrule
        Unlinkability & $S$ saw only $b_i$; $(s_i,\sigma_i)$ is new to it \\
        Unforgeability & one-more RSA (Bellare et al., 2003) \\
        No double spending & spent-serial set \\
        Price privacy & every resource costs one credit \\
        \bottomrule
      \end{tabular}
    \end{column}
    \begin{column}{0.22\textwidth}
      \centering
      \sprite[1.8cm]{mask}
    \end{column}
  \end{columns}
  \bigskip

  \textbf{Limits.} The batch purchase is linkable to $A$. Timing and IP are not hidden. The mint key is kept in memory.
\end{frame}

\begin{frame}{Spending policy}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      The policy runs in the x402 hook \texttt{onBeforePaymentCreation}: a refused payment is never signed.
      \begin{itemize}
        \item origin allowlist
        \item \$0.01 per call, \$0.05 per session
        \item my approval above \$0.004
      \end{itemize}
    \end{column}
    \begin{column}{0.47\textwidth}
      \centering
      \sprite[1.3cm]{shield}\hspace{3mm}\sprite[1cm]{stamp-denied}\hspace{2mm}\sprite[1cm]{stamp-approved}\\[2mm]
      \begin{term}[\linewidth]
        \cmd{policy 127.0.0.1/api/feeds/whale \$0.50}
        \out{DENY origin not on allowlist}
        \cmd{policy /credits/issue \$0.01}
        \out{NEEDS\_APPROVAL above \$0.004}
      \end{term}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Evaluation: is the purchased signal informative?}
  \begin{columns}[T]
    \begin{column}{0.62\textwidth}
      84 Gemini crypto ladders settled in the last 24 h, 736 contracts, priced 60 min before expiry.
      \medskip

      \small
      \begin{tabular}{@{}lcc@{}}
        \toprule
        Forecaster & all & $p\in(0.1,0.9)$ \\
        \midrule
        Lognormal, 2\,h realized vol. & \textbf{0.112} & \textbf{0.217} \\
        Spot past the strike & 0.152 & 0.342 \\
        Constant 0.5 & 0.250 & 0.250 \\
        \bottomrule
      \end{tabular}
    \end{column}
    \begin{column}{0.35\textwidth}
      \begin{term}[\linewidth]
        \cmd{bun backtest/calibration.ts}
        \out{events 84  contracts 736}
        \out{brier model  0.112}
        \out{brier naive  0.152}
      \end{term}
    \end{column}
  \end{columns}
  \medskip
  Brier score, lower is better. Overconfident above 80\% (0.97 predicted, 0.87 realized).
\end{frame}
